#include <iostream>
#include <string>
using namespace std;

void kaprekarNumbers(int p, int q) {
    int kp = 0;

    while (p <= q) {
        long long mul = (long long)p * p;

        string str = to_string(mul);
        int d = to_string(p).length();
        int len = str.length();

        string left = str.substr(0, len - d);
        string right = str.substr(len - d);

        int l = (left.empty()) ? 0 : stoi(left);
        int r = (right.empty()) ? 0 : stoi(right);

        int sum = l + r;
        if (sum == p) {
            cout << p << " ";
            kp = 1;
        }

        p++;
    }

    if (kp == 0) {
        cout << "INVALID RANGE";
    }
}
