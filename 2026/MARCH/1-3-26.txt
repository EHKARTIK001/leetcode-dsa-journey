THIS IS BRUTE FORCE APPROACH
class Solution {
public:
    int minPartitions(string n) {
        int count = 0 ; 
        while(true){
            bool az = true ; 
            for(int i = 0 ; i<n.size() ; i++){
                if(n[i]>'0'){
                    n[i]--;
                    az= false ; 

                }
            }
            if(az){
                break;
            }
            count++;
        }
        return count;
        
    }
};

// OPTIMIZED APPROACH (ONLY FIND MAX )
class Solution {
public:
    int minPartitions(string n) {
        int maxDigit = 0;
        for(char c : n) {
            maxDigit = max(maxDigit, c - '0');
        }
        return maxDigit;
    }
};
