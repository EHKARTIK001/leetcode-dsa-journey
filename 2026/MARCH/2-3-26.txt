#include <bits/stdc++.h>
using namespace std;
int main()
{
    int t ; 
    cin>>t;
    while(t--){
        long long n , k ;
        cin>>n>>k ; 
        vector<long long>a(n);
        for(int i = 0 ; i<n ;  i++){
            cin>>a[i];
        }
    // THIS IS HOW THE INPUT IS TAKEN 
    // THEN COPY THE ARRAY AND AFTER SORT AND CHECK 
    vector<long long >copy = a ; 
    sort(a.begin(),a.end());
    if(a==copy || k>1){
        cout<<"YES";
    }
    else{
        cout<<"NOT";
    }
}
    return 0 ;
}
