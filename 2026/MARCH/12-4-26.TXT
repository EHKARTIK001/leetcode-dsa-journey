#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

// Define the node structure
struct node {
    int data;
    node* left;
    node* right;
    node(int val) {
        data = val;
        left = NULL;
        right = NULL;
    }
};

// Function to build tree from preorder traversal
node* buildtree(vector<int>& preorder) {
    static int idx = 0;
    if (idx >= preorder.size() || preorder[idx] == -1) {
        idx++;
        return NULL;
    }
    node* root = new node(preorder[idx]);
    idx++;
    root->left = buildtree(preorder);
    root->right = buildtree(preorder);
    return root;
}

// THE MAIN LOGIC BEHIND THE HEIGHT OF POST ORDER TRAVERSAL LEFT RIGHT ROOT 
int height(node* root){
    if(root==NULL) return 0 ;
    int leftt = height(root->left);
    int rightt = height(root->right);
    return max(leftt , rightt)+1;
}
// THE LOGIC BEHIND THE NODE JUST ADD BOTH PLUS ROOT DATA 
int countnode(node* root){
    if(root==NULL){
        return 0 ; 
    }
    int leftt = countnode(root->left);
    int rightt = countnode(root->right);
    return leftt+rightt+1;

}
int sumnode(node* root){
    if(root==NULL){
        return 0 ; 
    }
    int leftt = sumnode(root->left);
    int rightt = sumnode(root->right);
    return leftt+rightt+root->data;
}
int main(){
vector<int> preorder ={1,1,-1,-1,3,4,-1,-1,5,-1,-1};
node* root = buildtree(preorder);
cout<<"HEIGHT : "<<height(root)<<endl;
cout<<"COUNT OF NODES : "<<countnode(root)<<endl;
cout<<"SUM OF NODES  : "<<sumnode(root)<<endl;
return 0;
}
