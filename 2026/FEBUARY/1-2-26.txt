class Solution {
public:
    int minimumCost(vector<int>& nums) {
        int n = nums.size();
        int score = nums[0];
        int ist  = INT_MAX ; 
            int snd = INT_MAX; 
        for(int i =1 ; i<n ; i++){
            if(nums[i]<ist){
                snd = ist;
                ist = nums[i];
            }
            else  if(nums[i]<snd){
                snd = nums[i];
            }
        }
        return score + ist + snd;
        
    }
};
