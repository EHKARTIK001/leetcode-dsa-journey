class Solution {
public:
    int minCost(string colors, vector<int>& neededTime) {
        int mtime = 0 ; 
         int mprev = 0 ; 
         int  n =colors.size();
         for(int i = 0 ; i<n ; i++){
            if(i>0 && colors[i]!=colors[i-1]){
                mprev = 0 ; 
            }
            int curr = neededTime[i];
            mtime+=min(curr , mprev);
            mprev = max(curr , mprev);
            
         } 
        return mtime ; 
    }
};
