#include<iostream>
#include<vector>
using namespace std ; 
class Node { // structure of node 
    public :
    int data ; 
    Node* next  ;

    Node(int val){
        data = val ;
        next = NULL;
    }

};
class queqe {
    Node* head ; 
     Node* tail; 
     public :
     queqe(){
        head = tail = NULL;
     }
     void pop(){
        if(empty()){
            cout<<" QUEUQ KHALI H ";
        }
        else {
            Node *temp ; 
            temp = head ;  
             head = head->next; 
             delete temp ;
        }

     }
     void push(int data ){
       Node* newNode = new Node(data); 
       if(empty()){
        head = tail =  newNode ; 
       }
       else{
       tail->next = newNode ; 
       tail = newNode ; 
     }
    }
     bool empty(){
         return head == NULL;

     }
     int front (){
        if(empty()){
            cout<<"KUCH NHI H ";
        }
        else {
            return head->data;
        }

     }

};
int main(){
    queqe q ; 
    q.push(1);
    q.push(2);
    q.push(3);
    while(!q.empty()){
        cout<<q.front()<<endl;
        q.pop();

    }
    cout<<endl ;
    return 0 ; 

}
